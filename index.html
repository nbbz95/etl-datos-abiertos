-- Inflación compuesta, devaluación del oficial y brecha promedio por año.
SELECT
    year(mes) AS anio,
    count(*) AS meses,
    round((exp(sum(ln(1 + inflacion_pct / 100))) - 1) * 100, 1) AS inflacion_anual_pct,
    round((exp(sum(ln(1 + coalesce(devaluacion_oficial_pct, 0) / 100))) - 1) * 100, 1) AS devaluacion_anual_pct,
    round(avg(brecha_pct), 1) AS brecha_prom_pct,
    round(avg(riesgo_pais_prom)) AS riesgo_pais_prom
FROM fct_mensual
WHERE oficial_prom IS NOT NULL
GROUP BY 1
ORDER BY 1 DESC;

-- ¿El dólar oficial corrió por encima o por debajo de la inflación? Últimos 12 meses.
WITH ult AS (
    SELECT * FROM fct_mensual WHERE devaluacion_oficial_pct IS NOT NULL ORDER BY mes DESC LIMIT 12
)
SELECT
    round((exp(sum(ln(1 + inflacion_pct / 100))) - 1) * 100, 1) AS inflacion_12m_pct,
    round((exp(sum(ln(1 + devaluacion_oficial_pct / 100))) - 1) * 100, 1) AS devaluacion_12m_pct,
    CASE
        WHEN sum(ln(1 + devaluacion_oficial_pct / 100)) < sum(ln(1 + inflacion_pct / 100))
        THEN 'El dólar subió menos que los precios (apreciación real del peso)'
        ELSE 'El dólar subió más que los precios (depreciación real del peso)'
    END AS lectura
FROM ult;

-- Correlación entre variables mensuales desde 2018.
SELECT
    round(corr(inflacion_pct, devaluacion_oficial_pct), 2) AS inflacion_vs_devaluacion,
    round(corr(inflacion_pct, brecha_pct), 2) AS inflacion_vs_brecha,
    round(corr(brecha_pct, riesgo_pais_prom), 2) AS brecha_vs_riesgo_pais
FROM fct_mensual
WHERE mes >= DATE '2018-01-01';
